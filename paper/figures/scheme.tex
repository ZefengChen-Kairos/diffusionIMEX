% ----------------------------------------------------------------- one stage
\node[hdr, below=11mm of split.south |- advbox.south, text width=12.8cm, align=flush center] (alh)
  {Algorithm: one stage of LSDIRK2 (Sec.~\ref{sec:time}), implicit step $\tau=a_{ii}\Delta t$};
\node[step, below=2mm of alh] (s0)
  {\textbf{0. Start of the time step} (thermal / chemical relaxation only)\\
   activation of the closures on $U^n$, projection onto the $pT$ / $pTg$ manifold;\\ closure type frozen over the stages};
\node[step, below=5mm of s0] (s1)
  {\textbf{1. Explicit advective update} on $U_{E,i}$, from $\tilde U_{I,i}$\\
   $X^*=X^s-r\,\delta_x\big[F^e[X]\big]$, \ $\alpha^*_1=\alpha^s_1-r\big(\delta_x[F^e[\alpha_1]]-\alpha^e_1\,\delta_x[\bar u^e]\big)$};
\node[step, below=5mm of s1] (s2)
  {\textbf{2. Frozen coefficients} on $X^*$\\
   faces: $\bar u^*_f$, $\bar\rho_f$, $Z$, weight $\varphi_f=(1-\tau c_f/\Delta x)^+$, factors $\psi_f$, velocity jump $D^*_f$\\
   cells: shock budget $Q_k=\lambda_k\rho\mathcal D$ from the shock relations (shooting)};
\node[step, below=5mm of s2] (s3)
  {\textbf{3. Newton iteration for the cell pressures $P$}\\[1pt]
   (a) $u_f=\bar u^*_f-\Theta_f(P_R-P_L)$, \ $p_f=W_fP_L+(1-W_f)P_R-\varpi_f$ \ (with implicit Laplace dissipation)\\
   (b) incremental fluxes $\Rightarrow$ $X^{\rm tr}$, strain $s$, $(\rho\mathbf u)^{\rm new}$, $E^{\rm new}$, $(\rho e)^C$\\
   (c) local closure in every cell $\Rightarrow$ $\bar p^{\rm cl}$ \\
   (d) $G(P)=\bar p^{\rm cl}-P=0$: Newton step (BiCGStab / Thomas, line search), repeat (a)--(c)};
\node[step, below=5mm of s3] (s4)
  {\textbf{4. Stage value}: $\rho\mathbf u$, $E$ from 3(b); $\alpha_k$, $m_k$, $\varepsilon_k$ from the closure; \ $U^{n+1}=U_2$};
\begin{scope}[on background layer]
  \node[outer, fit=(alh)(s0)(s1)(s2)(s3)(s4)] (algbox) {};
\end{scope}

% ----------------------------------------------------------------- local closure with energy partition
\node[step=5.8cm, fill=clgreen, draw=black!45, below left=12mm and 9mm of algbox.south, anchor=north east, minimum height=4.6cm] (cl)
  {\textbf{Local closure} (every residual evaluation)\\[2pt]
   $\alpha^{\rm tr}_k=\alpha^*_k(1-s)$, \ pressure work $\omega_kp_k\,s$\\[2pt]
   \textbf{energy defect}\\
   $\delta=(\rho e)^C-\sum_k\varepsilon^{\rm tr}_k+s\sum_k\omega_kp_k$\\[2pt]
   \textbf{partition} $\Lambda_k(\delta)$, $\sum_k\Lambda_k=\delta$:\\
   $0\le\delta\le Q$: shock share $Q_k/Q$\\
   excess over $Q$ and $\delta<0$: $\lambda^{\mathcal E}_k=\mathcal E_k/\sum_j\mathcal E_j$\\[2pt]
   $\Rightarrow\ \sum_k\varepsilon_k=(\rho e)^C$ exactly\\[2pt]
   relaxation closure $\Rightarrow$ $\Phi(p)=0$\\ $\bar p^{\rm cl}=\sum_k\alpha_kp_k$};

% ----------------------------------------------------------------- relaxation closures by rates
\node[step=5.8cm, fill=rlpurple, draw=black!45, below right=12mm and 9mm of algbox.south, anchor=north west, minimum height=4.6cm] (rl)
  {\textbf{Relaxation closure by rates}\\
   ordering: $\mu=\infty$ if $\theta>0$ or $\nu>0$\\ \phantom{ordering:} $\theta=\infty$ if $\nu>0$\\[3pt]
   \begin{tabular}{@{}l@{\ \ }l@{}}
   $\nu>0$:      & $\nu=\infty$: $pTg$, $T=T_{\rm sat}(p)$\\
                 & $\nu<\infty$: $Y$ from $\mathcal R(Y)=0$\\[2pt]
   $\theta>0$:   & $\theta=\infty$: $pT$, $T(p)$ explicit\\
                 & $\theta<\infty$: $p$ closure $+$ heat $\tau\mathcal Q(p)$\\[2pt]
   otherwise:    & $\mu=\infty$: $p$ closure, $\alpha_k(p)$\\
                 & $0<\mu<\infty$: $R$ from $\mathcal F(R)=0$\\
                 & $\mu=0$: frozen, $\alpha_k=\alpha^{\rm tr}_k/(1-s)$
   \end{tabular}\\[3pt]
   each case: one scalar equation\\ $\Phi(p)=0$ per cell};
