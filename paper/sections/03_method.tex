\subsubsection{Face velocity and face pressure}
% TODO: momentum-consistent pressure acceleration (half-cell momentum balance),

\begin{equation}
\begin{aligned}
u_f&=\bar u^*_f-\frac{\tau(P_{i+1}-P_i)}{\bar\rho_f\Delta x}-\varphi_f\,\psi^P_f\,\frac{P_{i+1}-P_i}{Z_i+Z_{i+1}},\\

\end{aligned}\qquad
\varphi_f=\Big(1-\frac{\tau\min(c_i,c_{i+1})}{\Delta x}\Big)^+ .
\label{eq:faces}
\end{equation}
\paragraph{Weight of the acoustic complement}
% Derivation in the text (B9.5 Thm 6):
Consider one implicit stage (backward Euler, step $\tau$) of linear acoustics about a state at rest
with sound speed $c$, written with the face quantities \eqref{eq:faces} and both complement terms
multiplied by the same weight $\varphi\ge0$. Expanding the amplification matrix for small wave numbers
gives, for both acoustic characteristic variables, the effective viscosity
\begin{equation}
D_{\rm stage}=\tfrac12c^2\tau+\varphi\,\tfrac12c\Delta x=\tfrac12c\Delta x\,(\nu+\varphi)+O(k^2),\qquad \nu=\frac{c\tau}{\Delta x},
\label{eq:dstage}
\end{equation}
the first term coming from backward Euler and the second from the complement.
% TODO: one line on the expansion (or the modified equation) and the numerical check.
The three-point semi-discretisation $\dot w_i=-\frac{c}{2\Delta x}(w_{i+1}-w_{i-1})+\frac{D}{\Delta x^2}\delta^2w_i$
has non-negative coefficients if and only if $D\ge\tfrac12c\Delta x$, the upwind (Godunov) viscosity.
Requiring that every stage carries at least this viscosity on resolved acoustic waves, and no more,
\[
D_{\rm stage}=\max\big(\tfrac12c\Delta x,\ \tfrac12c^2\tau\big),
\]
fixes the weight uniquely: $\varphi=1-\nu$ for $\nu<1$, and $\varphi=0$ for $\nu\ge1$, where backward
Euler alone already supplies $\tfrac12c^2\tau\ge\tfrac12c\Delta x$. Hence $\varphi=(1-\nu)^+$ in
\eqref{eq:faces}. The two terms are the two halves of the Godunov dissipation on the two characteristic
variables and therefore share the weight.

\paragraph{Second-order factors}
The complement \eqref{eq:faces} is the acoustic Godunov dissipation evaluated on the cell values
$(P_i,u_i)$, $(P_{i+1},u_{i+1})$, i.e.\ it is first order. To make it second order where the solution is
smooth we evaluate it on the reconstructed face states of the explicit operator (Sec.~\ref{sec:explicit}),
which are the states the same face carries. Write $D=P_{i+1}-P_i$ and let $\sigma_i$ be the slope of $P$ in
cell $i$; the jump of the reconstructed face pressures is
\begin{equation}
P^R-P^L=D-\tfrac12(\sigma_i+\sigma_{i+1})=:\psi_f D .
\label{eq:psi_def}
\end{equation}
The pressure is not reconstructed by the scheme; it is a function of the reconstructed volume-transported
quantities $X\in\{\alpha_k,m_k,\varepsilon_k\}$ and of the velocity. These are reconstructed with the
shared limiter, $X^{L,R}=X_i\pm\tfrac12\phi_i\,\delta^cX_i$ with $\phi_i\in[0,1]$ the largest factor for
which every slope is a minmod slope, $\phi_i=0$ at a local extremum of any of them, and the velocity
separately with its own minmod slope (Sec.~\ref{sec:explicit}). Linearising $P(Q^R)-P(Q^L)$ about the
cell states gives
\[
P(Q^R)-P(Q^L)=D-\tfrac12\phi_s\big(\delta^cP_i+\delta^cP_{i+1}\big)+O(\Delta x^3),\qquad \phi_s=\min(\phi_i,\phi_{i+1}),
\]
so the pressure slope of the scheme's own face states is the central slope scaled by the shared factor
(in $\phi_i$ we also include the smoothness ratio of the velocity, since the face state $Q$ contains $u$).
Two requirements fix $\sigma$:
\begin{enumerate}
\item the dissipation must not exceed the scheme's own reconstruction, $\sigma_i=\phi_s\,\cdot$(slope of $P$),
so that the pressure jump is that of the face states carrying the transported quantities;
\item the dissipation must be LED for $P$ itself (Jameson 1995): its coefficient must be non-negative and
equal to the first-order value at a local extremum of $P$, which holds if the slope of $P$ is a minmod
slope, $\sigma_i=\phi_s\,\sigma^{\rm mm}_i[P]$.
\end{enumerate}
Inserting $\sigma_i=\phi_s\sigma^{\rm mm}_i[P]$ into \eqref{eq:psi_def} gives
\begin{equation}
\psi^P_f=1-\phi_s\big(1-\psi^{\rm mm}_f[P]\big),\qquad
\psi^{\rm mm}_f[P]=1-\frac{\sigma^{\rm mm}_i[P]+\sigma^{\rm mm}_{i+1}[P]}{2D}\in[0,1].
\label{eq:psiP}
\end{equation}
The velocity-jump term modifies only the momentum flux and acts on the separately reconstructed velocity;
the same two requirements with the velocity's own minmod slope give
\begin{equation}
\psi^u_f=\psi^{\rm mm}_f[u].
\label{eq:psiu}
\end{equation}
Consequently: (i) $0\le\psi\le1$, so both dissipation coefficients in \eqref{eq:faces} stay non-negative;
(ii) $\psi^P_f=1$ (first order, the full Godunov dissipation) at a local extremum of $P$ or of any
volume-transported quantity, in particular at a contact, where $P$ and $u$ are continuous but the
impedance jumps; $\psi^u_f=1$ at a local extremum of $u$; (iii) where everything is smooth,
$\psi=O(\Delta x)$ and the complement is second order. This is the LED criterion of Jameson (1995) for a
limited dissipation: first order at an extremum of every quantity carried by the flux that the term
modifies, second order otherwise.

\subsection{Asymptotic limits}
\label{sec:limits}
\begin{theorem}[Relaxation limits]\label{thm:limits}
% TODO (B7 Thms 1 and 2 merged):
% (i) rates -> infinity: O(1/(tau mu)), O(1/(tau theta)), O(1/(tau nu)) convergence to the
%     equilibrium closures; the limit scheme is a consistent discretisation of the
%     equilibrium model (Kapila volume equation);
% (ii) rates -> 0: the frozen scheme is recovered exactly.
\end{theorem}
\begin{proposition}[Formal low-Mach limit]\label{prop:lowmach}
Non-dimensionalise pressure by $\rho_rc_r^2$, keep $\Delta t$ and $\Delta x$ fixed, and take
well-prepared data $P=P_0+M^2P_2$ with $P_0$ constant and $\mathbf u=O(1)$. Then, formally as $M\to0$,
every implicit stage satisfies
\begin{enumerate}
\item $(\delta\cdot u_f)_i=O(M^2)$: the face velocities are discretely divergence-free;
\item $\delta\cdot\big(\bar u^*_f-\tau\,\delta P_2/(\bar\rho_f\Delta x)\big)=O(M^2)$: $P_2$ solves the discrete
Poisson equation of a face-velocity projection;
\item the cell velocities are corrected by the density-weighted face average of $P_2$ ($P_0$ cancels),
an approximate projection;
\item the acoustic complement vanishes ($\varphi_f=0$ once $M<a_{ii}\,\mathrm{CFL}$), the explicit part
carries only the material-speed Rusanov dissipation, and the closure returns equilibrium states, all
independently of $M$.
\end{enumerate}
The limit scheme is therefore a consistent discretisation of the incompressible mixture Euler equations
(explicit convection plus a face-velocity projection), with a time step independent of $M$.
\end{proposition}
